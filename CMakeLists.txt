cmake_minimum_required(VERSION 3.21)
project(image_codec CXX)
add_compile_definitions(_USE_MATH_DEFINES NOMINMAX)
set(CMAKE_CXX_STANDARD 17)
add_executable(llrice-c encoder.cpp llrice.hpp bitstream.hpp rlgr.hpp pool.hpp polyfill.hpp rice.hpp ring.hpp profiling.hpp image.hpp)
add_executable(llrice-d decoder.cpp llrice.hpp bitstream.hpp rlgr.hpp pool.hpp polyfill.hpp rice.hpp ring.hpp profiling.hpp image.hpp)
target_compile_options(llrice-c PRIVATE -mcrc32 -msse4.1)
target_compile_options(llrice-d PRIVATE -mcrc32 -msse4.1)
